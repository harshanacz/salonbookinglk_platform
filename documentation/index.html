Part 02: Top Rated Services

Logic

Services are ranked using a combined score of rating and review count:

// Top Rated Services Algorithm
const topRatedServices = [...transformedServices]
    .sort((a, b) => {
        const scoreA = a.rating + (a.reviews / 100);
        const scoreB = b.rating + (b.reviews / 100);
        return scoreB - scoreA;
    })
    .slice(0, 8);

// Increment service view count
export async function incrementServiceViews(serviceId: string): Promise<number> {
    const key = `service:${serviceId}:views`;
    const newViews = await redis.incr(key);
    
    // Set expiration for 30 days
    await redis.expire(key, 30 * 24 * 60 * 60);
    
    return newViews;
}

// Get service view count
export async function getServiceViews(serviceId: string): Promise<number> {
    const key = `service:${serviceId}:views`;
    const views = await redis.get(key) || 0;
    return Number(views);
}

Scheduled Sync to MongoDB

// lib/scheduledTasks.ts
import { redis } from './redis';
import { connectToDatabase } from './mongodb';
import Service from '@/models/Service';

export async function syncRedisViewsToMongoDB() {
    try {
        await connectToDatabase();
        
        // Get all service view keys from Redis
        const keys = await redis.keys('service:*:views');
        
        for (const key of keys) {
            const serviceId = key.split(':')[1];
            const views = await redis.get(key);
            
            if (views && Number(views) > 0) {
                await Service.findByIdAndUpdate(serviceId, {
                    $inc: { views: Number(views) }
                });
                
                // Reset Redis counter after sync
                await redis.del(key);
            }
        }
        
        console.log('✅ Redis views synced to MongoDB');
    } catch (error) {
        console.error('❌ Error syncing Redis views:', error);
    }
}

// app/api/ratings/route.ts
export async function POST(req: Request) {
    try {
        const { 
            appointmentId, 
            serviceId, 
            salonId, 
            serviceRating, 
            salonRating, 
            comment, 
            userId 
        } = await req.json();
        
        await connectToDatabase();
        
        // Validate appointment belongs to user
        const appointment = await Appointment.findOne({
            _id: appointmentId,
            userId: userId
        });
        
        if (!appointment) {
            return NextResponse.json({ error: 'Appointment not found' }, { status: 404 });
        }
        
        // Create rating record
        const rating = new Rating({
            appointmentId,
            serviceId,
            salonId,
            userId,
            serviceRating,
            salonRating,
            comment,
            createdAt: new Date()
        });
        
        await rating.save();
        
        // Update service average rating
        await updateServiceRating(serviceId, serviceRating);
        
        // Update salon average rating
        await updateSalonRating(salonId, salonRating);
        
        // Mark appointment as reviewed
        await Appointment.findByIdAndUpdate(appointmentId, {
            hasReview: true,
            reviewId: rating._id
        });
        
        return NextResponse.json({ 
            success: true, 
            ratingId: rating._id 
        });
    } catch (error) {
        console.error('[RATING_POST]', error);
        return NextResponse.json({ error: 'Failed to submit rating' }, { status: 500 });
    }
}

async function updateServiceRating(serviceId: string, newRating: number) {
    const service = await Service.findById(serviceId);
    if (!service) return;
    
    const currentAverage = service.averageRating || 0;
    const currentCount = service.reviewCount || 0;
    
    const newAverage = (currentAverage * currentCount + newRating) / (currentCount + 1);
    
    await Service.findByIdAndUpdate(serviceId, {
        averageRating: Math.round(newAverage * 10) / 10,
        reviewCount: currentCount + 1
    });
}

async function updateSalonRating(salonId: string, newRating: number) {
    const salonContext = await SalonContext.findOne({ userId: salonId });
    if (!salonContext) return;
    
    const currentAverage = salonContext.averageRating || 0;
    const currentCount = salonContext.totalReviews || 0;
    
    const newAverage = (currentAverage * currentCount + newRating) / (currentCount + 1);
    
    await SalonContext.findOneAndUpdate(
        { userId: salonId },
        {
            averageRating: Math.round(newAverage * 10) / 10,
            totalReviews: currentCount + 1
        }
    );
}

// app/salon-owner/services/create/page.tsx
export default function CreateServicePage() {
    const [currentStep, setCurrentStep] = useState(1);
    const [formData, setFormData] = useState({});
    const [imageUrls, setImageUrls] = useState([]);
    const [isLoading, setIsLoading] = useState(false);
    
    const totalSteps = 4;
    const isLastStep = currentStep === totalSteps;
    
    // Step validation logic
    const isStepValid = () => {
        if (currentStep === 1) {
            return formData.title && formData.type;
        }
        if (currentStep === 2) {
            return formData.duration && formData.price;
        }
        if (currentStep === 3) {
            return formData.staffAssigned && formData.staffAssigned.length > 0;
        }
        if (currentStep === 4) {
            return imageUrls.length > 0;
        }
        return false;
    };
    
    const handleNext = () => {
        if (currentStep < totalSteps) {
            setCurrentStep(prev => prev + 1);
        }
    };
    
    const handleSubmit = async () => {
        setIsLoading(true);
        try {
            const serviceData = {
                ...formData,
                images: imageUrls,
                status: 'PENDING' // Requires admin approval
            };
            
            const response = await fetch('/api/salon-owner/services', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(serviceData)
            });
            
            if (response.ok) {
                setShowSuccessDialog(true);
            }
        } catch (error) {
            console.error('Error creating service:', error);
        } finally {
            setIsLoading(false);
        }
    };
    
    return (
        <div className="create-service-page">
            <TabNavigation currentStep={currentStep} totalSteps={totalSteps} />
            
            {currentStep === 1 && (
                <BasicDetailsForm 
                    formData={formData} 
                    setFormData={setFormData} 
                />
            )}
            {currentStep === 2 && (
                <PricingForm 
                    formData={formData} 
                    setFormData={setFormData} 
                />
            )}
            {currentStep === 3 && (
                <StaffForm 
                    formData={formData} 
                    setFormData={setFormData} 
                />
            )}
            {currentStep === 4 && (
                <ServiceImages 
                    imageUrls={imageUrls} 
                    setImageUrls={setImageUrls} 
                />
            )}
            
            <div className="form-actions">
                {currentStep > 1 && (
                    <Button onClick={() => setCurrentStep(prev => prev - 1)}>
                        Previous
                    </Button>
                )}
                
                {!isLastStep ? (
                    <Button 
                        onClick={handleNext} 
                        disabled={!isStepValid()}
                    >
                        Next
                    </Button>
                ) : (
                    <Button 
                        onClick={handleSubmit} 
                        disabled={!isStepValid() || isLoading}
                    >
                        {isLoading ? 'Creating...' : 'Create Service'}
                    </Button>
                )}
            </div>
        </div>
    );
}

export async function PUT(req: Request, { params }: { params: { id: string } }) {
    try {
        const { userId } = auth();
        if (!userId) {
            return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
        }
        
        const updates = await req.json();
        
        await connectToDatabase();
        
        // Verify service ownership
        const service = await Service.findOne({ _id: params.id, userId });
        if (!service) {
            return NextResponse.json({ error: 'Service not found' }, { status: 404 });
        }
        
        // If service was previously approved and now being edited, 
        // set status back to pending
        if (service.status === 'ACTIVE') {
            updates.status = 'PENDING';
        }
        
        const updatedService = await Service.findByIdAndUpdate(
            params.id,
            { ...updates, updatedAt: new Date() },
            { new: true }
        );
        
        return NextResponse.json({ service: updatedService });
    } catch (error) {
        return NextResponse.json({ error: 'Failed to update service' }, { status: 500 });
    }
}

export async function DELETE(req: Request, { params }: { params: { id: string } }) {
    try {
        const { userId } = auth();
        if (!userId) {
            return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
        }
        
        await connectToDatabase();
        
        // Verify service ownership and delete
        const service = await Service.findOneAndDelete({ _id: params.id, userId });
        if (!service) {
            return NextResponse.json({ error: 'Service not found' }, { status: 404 });
        }
        
        return NextResponse.json({ success: true });
    } catch (error) {
        return NextResponse.json({ error: 'Failed to delete service' }, { status: 500 });
    }
}

                setHasAccess(false);

        };

    }, [user, isLoaded]);

    if (!hasAccess) {
        return null;
    }
    
    return (

"use client";

import Head from 'next/head';

interface SEOHeadProps {
    title: string;
    description: string;
    keywords?: string;
    image?: string;
    url?: string;
    type?: 'website' | 'article';
    structuredData?: object;
}

export default function SEOHead({
    title,
    description,
    keywords,
    image = '/og-image.jpg',
    url = '',
    type = 'website',
    structuredData
}: SEOHeadProps) {
    const fullUrl = `https://salonbooking.lk${url}`;
    const fullTitle = title.includes('SalonBooking.lk') ? title : `${title} | SalonBooking.lk`;
    
    return (
        <Head>
            <title>{fullTitle}</title>
            <meta name="description" content={description} />
            {keywords && <meta name="keywords" content={keywords} />}
            <link rel="canonical" href={fullUrl} />
            
            {/* Open Graph */}
            <meta property="og:title" content={fullTitle} />
            <meta property="og:description" content={description} />
            <meta property="og:type" content={type} />
            <meta property="og:url" content={fullUrl} />
            <meta property="og:image" content={`https://salonbooking.lk${image}`} />
            <meta property="og:site_name" content="SalonBooking.lk" />
            
            {/* Twitter */}
            <meta name="twitter:card" content="summary_large_image" />
            <meta name="twitter:title" content={fullTitle} />
            <meta name="twitter:description" content={description} />
            <meta name="twitter:image" content={`https://salonbooking.lk${image}`} />
            <meta name="twitter:creator" content="@salonbookinglk" />
            
            {/* Structured Data */}
            {structuredData && (
                <script
                    type="application/ld+json"
                    dangerouslySetInnerHTML={{
                        __html: JSON.stringify(structuredData)
                    }}
                />
            )}
        </Head>
    );
}
